-- GENE table
CREATE TABLE GENE (
    gene_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    gene_symbol VARCHAR2(50) UNIQUE,
    gene_name VARCHAR2(255),
    entrez_id VARCHAR2(20)
);

-- STUDY table
CREATE TABLE STUDY (
    study_id NUMBER PRIMARY KEY,
    title VARCHAR2(1000),
    pmid VARCHAR2(20) UNIQUE,
    year NUMBER,
    journal VARCHAR2(255),
    impact_factor NUMBER
);

-- GENE_RADIATION_LINK
CREATE TABLE GENE_RADIATION_LINK (
    link_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    gene_id NUMBER REFERENCES GENE(gene_id),
    study_id NUMBER REFERENCES STUDY(study_id),
    evidence_type VARCHAR2(50),
    effect_direction VARCHAR2(50),
    confidence_score NUMBER
);

-- CANCER_TYPE
CREATE TABLE CANCER_TYPE (
    cancer_id NUMBER PRIMARY KEY,
    cancer_name VARCHAR2(100)
);

-- GENE_CANCER_LINK
CREATE TABLE GENE_CANCER_LINK (
    gene_cancer_link_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    gene_id NUMBER REFERENCES GENE(gene_id),
    cancer_id NUMBER REFERENCES CANCER_TYPE(cancer_id)
);
